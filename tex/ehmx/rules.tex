\section{solver の契約}
\[
\begin{gathered}
S:\mathcal C_X\times\mathcal C_X\longrightarrow
 \mathsf{solved}(R,\theta)\mid\mathsf{unresolved}(R)\mid\mathsf{inconsistent},\\
[\theta]=\bigwedge_{i\in\dom_T\theta}(\alpha_i=\theta_T(i))
 \land\bigwedge_{j\in\dom_E\theta}(\mu_j=\theta_E(j)),\\
\Solved(\theta)\iff\begin{cases}
\forall i\in\dom_T\theta.\ \dom\theta\cap\supp(\theta_T(i))=\varnothing,\\
\forall j\in\dom_E\theta.\ \dom_E\theta\cap\FV_E(\theta_E(j))=\varnothing.
\end{cases}
\end{gathered}
\]
\textbf{部分 solver}（S1・S2・P1 型の契約 \cite{OutsideIn}）
\[
S(G,W)=\mathsf{solved}(R,\theta)\Longrightarrow
\begin{cases}
 \Solved(\theta),\\
 \dom\theta\cap\FV(G,R)=\varnothing,\\
 G\land R\models_X\theta W,\\
 G\land W\models_X R\land[\theta],\\
 \Sat(G\land R).
\end{cases}
\]
\[
\begin{aligned}
S(G,W)=\mathsf{unresolved}(R)&\Rightarrow G\land R\equiv_X G\land W,\\
S(G,W)=\mathsf{inconsistent}&\Rightarrow\neg\Sat(G\land W).
\end{aligned}
\]
\textbf{完全性の追加契約}
\[
\boxed{\operatorname{Complete}(S)\iff
 \forall G,W.\ \Sat(G\land W)\Rightarrow\exists R,\theta.\ S(G,W)=\mathsf{solved}(R,\theta)}.
\]
$W$ は handler の量化制約も含む．$R=\top$ は要求しない．
\begin{lemma}[解集合の保存]
\[
 S(G,W)=\mathsf{solved}(R,\theta)
 \Longrightarrow G\land W\equiv_X G\land R\land[\theta].
\]
\end{lemma}
\begin{proof}
\[
\begin{aligned}
\rho\models G\land W&\Rightarrow\rho\models R\land[\theta],\\
\rho\models G\land R\land[\theta]
 &\Rightarrow\theta^*\rho\models W\ \land\ \theta^*\rho=\rho
 \Rightarrow\rho\models W.
\end{aligned}
\]
\end{proof}
\textbf{契約のモデル}
\[
\begin{aligned}
S_{?}(G,W)&=\mathsf{unresolved}(W),\\
S_{\mathrm{sem}}(G,W)&=
 \begin{cases}\mathsf{solved}(W,\mathrm{id})&\Sat(G\land W),\\
 \mathsf{inconsistent}&\neg\Sat(G\land W).
 \end{cases}
\end{aligned}
\]
\[
\operatorname{PartialSolver}(S_?)\qquad
\operatorname{PartialSolver}(S_{\mathrm{sem}})\land\operatorname{Complete}(S_{\mathrm{sem}}).
\]
$S_{\mathrm{sem}}$ は非計算的なモデル．実行可能な判定手続きは別の実装義務．
HM(X) の制約形式と型項再構成の区別は \cite[\S4.4]{Sulzmann2000}．

\section{生成と合成の定理}
この節の健全性では $\Closed(\Sigma)$ を仮定する．
\begin{theorem}[固定した生成算法]
\[
\begin{gathered}
P(\Gamma,t,\epsilon)\subseteq F,\quad\Gen(e,\Gamma,t,\epsilon;F)=D\\
\Longrightarrow\quad\top;\Gamma\vdash_N e:t!\epsilon\Rightarrow D,\\[1mm]
\exists\rho.\ \ground\Gamma_\rho\vdash_g e:\eval t_\rho!\eval\epsilon_\rho\\
\Longrightarrow\quad\exists D.\ \Gen(e,\Gamma,t,\epsilon;F)=D\ \land\\
\forall\rho.\ \ground\Gamma_\rho\vdash_g e:\eval t_\rho!\eval\epsilon_\rho
 \Rightarrow\exists\nu=_{F}\rho.\ \nu\models D.
\end{gathered}
\]
\end{theorem}
\begin{proof}
$e$ に関する構造帰納法．
\[
\begin{aligned}
\lambda,\mathrm{app},\mathrm{let},\mathrm{perform}:&\quad
 \rho\xmapsto{\mathrm{fresh}(F)}\rho_0\xmapsto{\mathrm{IH}_1}\rho_1
 \xmapsto{\mathrm{IH}_2}\rho_2,\\
 &\rho_{i+1}=_{F\cup\supp D_1\cup\cdots\cup\supp D_i}\rho_i;\\
\mathrm{handler}:&\quad
 \forall\rho_B.\ \exists\rho_J.\ \rho_J\models D_o\land t_r\leq t_r\land\epsilon_o\subseteq\chi,\\
 &J\cap F_o=\varnothing,\quad B\cap K_o=\varnothing,
 \quad\Scope(B,J;H,\ldots).
\end{aligned}
\]
\end{proof}
\begin{theorem}[射影後の正確性]
\[
\Proj(\Gamma,e,t,\epsilon)=D\land\GReal(\Gamma,\rho)
\Longrightarrow
\bigl(\rho\models D\iff\ground\Gamma_\rho\vdash_g e:\eval t_\rho!\eval\epsilon_\rho\bigr).
\]
\end{theorem}
\begin{proof}
$(\Rightarrow)$：生成健全性と導出の意味解釈．
$(\Leftarrow)$：$\nu=_{F}\rho$ を前定理で得て，
$\omega=\rho[\supp D_{\mathrm{raw}}\setminus F\mapsto\nu]$ を存在証人とする．
\end{proof}
\begin{theorem}[成功時の健全性]
\[
\begin{gathered}
\Inf^{\mathrm{at}}_{N,S}(\Gamma,e,t,\epsilon)=\mathsf{solved}(R,\theta),\quad
\rho\models R,\quad\GReal(\Gamma,\theta^*\rho)\\
\Longrightarrow\quad
\ground\Gamma_{\theta^*\rho}\vdash_g e:\eval{\theta t}_\rho!\eval{\theta\epsilon}_\rho.
\end{gathered}
\]
\[
\Gamma'=\theta\Gamma,\quad\Real_\top(\Gamma),\quad\Adm_{\Sat}(\Gamma')
\quad\Longrightarrow\quad
\operatorname{Typing}_{\Sat}(R,\Gamma',e,\theta t,\theta\epsilon).
\]
\end{theorem}
\begin{proof}
$\rho\models R\Rightarrow\theta^*\rho\models D\Rightarrow
\ground\Gamma_{\theta^*\rho}\vdash_g e:\eval t_{\theta^*\rho}!\eval\epsilon_{\theta^*\rho}$．
$\Sat(R)$ と \key{Typing.of\_ground} により再構成する．
\end{proof}
\begin{theorem}[solver の契約を仮定した完全性]
\[
\boxed{\operatorname{Complete}(S)\land\Real_C(\Gamma)\land
 \operatorname{Typing}_{\Sat}(C,\Gamma,e,\sigma,\epsilon)
 \Rightarrow\exists R,\theta.\ \Inf_{N,S}(\Gamma,e)=\mathsf{solved}(R,\theta)}.
\]
\end{theorem}
\begin{proof}
$\rho\models C\Rightarrow\ground\Gamma_\rho\vdash_g e:u!E$．
新鮮な結果名に $(u,E)$ を割り当てる．生成完全性より $\Sat(D)$，
$\operatorname{Complete}(S)$ より $S(\top,D)=\mathsf{solved}(R,\theta)$．
\end{proof}

\section{実際の出力の主要性}
\[
\frac{A\cap\FV(\Gamma)=\varnothing\quad\Adm_{\Sat}(\exists A.C)\quad
 \operatorname{Typing}_{\Sat}(C,\Gamma,e,t,\epsilon)}
 {\operatorname{ComputationTyping}_{\Sat}(\exists A.C,\Gamma,e,\forall A.C\Rightarrow(t!\epsilon))}
\]
$\operatorname{ComputationTyping}$ は許容される制約への強化で閉じる．

\[
\begin{aligned}
\operatorname{generalize}(\Gamma,R,t,\epsilon)&=\forall A.R\Rightarrow(t!\epsilon),\\
A&=(\supp R\cup\supp t\cup(\varnothing,\FV_E\epsilon))\setminus\supp\Gamma,\\
s&=\forall A.R\Rightarrow(t!\epsilon),\\
\Fin(s;C,u,\eta)&\iff\exists\phi\text{（有限）}.\ \dom\phi\subseteq A\ \land\\
 &\qquad\forall\rho\models C.\ \phi^*\rho\models R\land
 \eval t_{\phi^*\rho}\leq\eval u_\rho\land\eval\epsilon_{\phi^*\rho}=\eval\eta_\rho.
\end{aligned}
\]
\begin{theorem}[開環境の有限因子]
\[
\begin{gathered}
\Inf^{\mathrm{at}}_{N,S}(\Gamma,e,t,\epsilon)=\mathsf{solved}(R,\theta),\qquad
\Gamma'=\sigma\Gamma,\quad\Real_C(\Gamma'),\\
\operatorname{Typing}_{\Sat}(C,\Gamma',e,\sigma t,\sigma\epsilon)\\
\Longrightarrow\quad
\exists\phi\text{（有限）}.\ \forall\rho\models C.\
 \phi^*\rho\models R\land\theta^*(\phi^*\rho)=\sigma^*\rho.
\end{gathered}
\]
\end{theorem}
\begin{proof}
$\phi=\sigma$ とする．
\[
\rho\models C\Rightarrow\sigma^*\rho\models D
 \Rightarrow\sigma^*\rho\models R\land[\theta]
 \Rightarrow\theta^*(\sigma^*\rho)=\sigma^*\rho.
\]
\end{proof}
\begin{theorem}[閉環境の主要計算型・主要値型]
\[
\begin{gathered}
\Closed(\Sigma),\quad\supp\Gamma=\varnothing,\quad
\Real_\top(\Gamma),\quad\Adm_{\Sat}(\Gamma),\\
\Inf_{N,S}(\Gamma,e)=\mathsf{solved}(R,\theta),\quad
p=f_T^{\supp\Gamma}(0),\quad q=f_E^{\supp\Gamma}(0),\\
s=\operatorname{generalize}(\Gamma,R,\theta\alpha_p,\theta\mu_q)\\
\Longrightarrow\quad
\operatorname{ComputationTyping}_{\Sat}(\top,\Gamma,e,s)\ \land\\
\forall C,t,\epsilon.\ \operatorname{Typing}_{\Sat}(C,\Gamma,e,t,\epsilon)
 \Rightarrow\Fin(s;C,t,\epsilon).
\end{gathered}
\]
$s=\forall A.R\Rightarrow(u!\eta)$ と書く．値なら $(\top,\forall A.R\Rightarrow u)$ は主要対で，即時効果は空．
\end{theorem}
\begin{proof}
生成制約を $D$ とする．$\Sat(R)\land R\models\theta D\Rightarrow\Sat(D)$．
生成の主要性と実行結果の一意性から，実際の $D$ の有限被覆を得る．
\[
\begin{aligned}
\rho\models C&\Rightarrow\sigma^*\rho\models D
 \land\eval{\alpha_p}_{\sigma^*\rho}\leq\eval t_\rho
 \land\eval{\mu_q}_{\sigma^*\rho}=\eval\epsilon_\rho,\\
D\equiv_X R\land[\theta]&\Rightarrow
 \sigma^*\rho\models R\land\theta^*(\sigma^*\rho)=\sigma^*\rho.
\end{aligned}
\]
$\sigma$ を $s$ の束縛名へ制限し，同じ有限因子を全ての $\rho\models C$ に用いる．
値の場合は \key{ComputationTyping.}\allowbreak\key{principal\_value} を適用する．
\end{proof}
\[
\begin{aligned}
\Inf^{\mathrm{at}}_{N,S}(\Gamma,e,t,\epsilon)=\mathsf{inconsistent}
 &\Rightarrow\neg\exists\rho.\ \ground\Gamma_\rho\vdash_g e:\eval t_\rho!\eval\epsilon_\rho,\\
\Inf^{\mathrm{at}}_{N,S}(\Gamma,e,t,\epsilon)=\mathsf{unresolved}(R)
 &\Rightarrow\exists D.\ \Proj(\Gamma,e,t,\epsilon)=D\land R\equiv_X D.
\end{aligned}
\]
