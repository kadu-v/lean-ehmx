\section{solver の契約}
\[
\begin{gathered}
S:\mathcal C_X\times\mathcal C_X\longrightarrow
 \mathsf{solved}(R,\theta)\mid\mathsf{unresolved}(R)\mid\mathsf{inconsistent},\\
[\theta]=\bigwedge_{i\in\dom_T\theta}(\alpha_i=\theta_T(i))
 \land\bigwedge_{j\in\dom_E\theta}(\mu_j=\theta_E(j)),\\
\Solved(\theta)\iff\begin{cases}
\forall i\in\dom_T\theta.\ \dom\theta\cap\supp(\theta_T(i))=\varnothing,\\
\forall j\in\dom_E\theta.\ \dom_E\theta\cap\FV_E(\theta_E(j))=\varnothing.
\end{cases}
\end{gathered}
\]
\textbf{部分 solver}（S1・S2・P1 型の契約 \cite{OutsideIn}）
\[
S(G,W)=\mathsf{solved}(R,\theta)\Longrightarrow
\begin{cases}
 \Solved(\theta),\\
 \dom\theta\cap\FV(G,R)=\varnothing,\\
 G\land R\models_X\theta W,\\
 G\land W\models_X R\land[\theta],\\
 \Sat(G\land R).
\end{cases}
\]
\[
\begin{aligned}
S(G,W)=\mathsf{unresolved}(R)&\Rightarrow G\land R\equiv_X G\land W,\\
S(G,W)=\mathsf{inconsistent}&\Rightarrow\neg\Sat(G\land W).
\end{aligned}
\]
\textbf{完全性の追加契約}
\[
\boxed{\operatorname{Complete}(S)\iff
 \forall G,W.\ \Sat(G\land W)\Rightarrow\exists R,\theta.\ S(G,W)=\mathsf{solved}(R,\theta)}.
\]
$W$ は handler の量化制約も含む．$R=\top$ は要求しない．
\begin{lemma}[解集合の保存]
\[
 S(G,W)=\mathsf{solved}(R,\theta)
 \Longrightarrow G\land W\equiv_X G\land R\land[\theta].
\]
\end{lemma}
\begin{proof}
\[
\begin{aligned}
\rho\models G\land W&\Rightarrow\rho\models R\land[\theta],\\
\rho\models G\land R\land[\theta]
 &\Rightarrow\theta^*\rho\models W\ \land\ \theta^*\rho=\rho
 \Rightarrow\rho\models W.
\end{aligned}
\]
\end{proof}
\textbf{契約のモデル}
\[
\begin{aligned}
S_{?}(G,W)&=\mathsf{unresolved}(W),\\
S_{\mathrm{sem}}(G,W)&=
 \begin{cases}\mathsf{solved}(W,\mathrm{id})&\Sat(G\land W),\\
 \mathsf{inconsistent}&\neg\Sat(G\land W).
 \end{cases}
\end{aligned}
\]
\[
\operatorname{PartialSolver}(S_?)\qquad
\operatorname{PartialSolver}(S_{\mathrm{sem}})\land\operatorname{Complete}(S_{\mathrm{sem}}).
\]
$S_{\mathrm{sem}}$ は非計算的なモデル．計算的構成は後述の $\operatorname{SatDecider}(X)$ を用いる．

\section{生成と合成の定理}
この節の健全性では $\Closed(\Sigma)$ を仮定する．
\begin{theorem}[固定した生成算法]
\[
\begin{gathered}
P(\Gamma,t,\epsilon)\subseteq F,\quad\Gen(e,\Gamma,t,\epsilon;F)=D\\
\Longrightarrow\quad\top;\Gamma\vdash_N e:t!\epsilon\Rightarrow D,\\[1mm]
\exists\rho.\ \ground\Gamma_\rho\vdash_g e:\eval t_\rho!\eval\epsilon_\rho\\
\Longrightarrow\quad\exists D.\ \Gen(e,\Gamma,t,\epsilon;F)=D\ \land\\
\forall\rho.\ \ground\Gamma_\rho\vdash_g e:\eval t_\rho!\eval\epsilon_\rho
 \Rightarrow\exists\nu=_{F}\rho.\ \nu\models D.
\end{gathered}
\]
\end{theorem}
\begin{proof}
$e$ に関する構造帰納法．
\[
\begin{aligned}
\lambda,\mathrm{app},\mathrm{let},\mathrm{perform}:&\quad
 \rho\xmapsto{\mathrm{fresh}(F)}\rho_0\xmapsto{\mathrm{IH}_1}\rho_1
 \xmapsto{\mathrm{IH}_2}\rho_2,\\
 &\rho_{i+1}=_{F\cup\supp D_1\cup\cdots\cup\supp D_i}\rho_i;\\
\mathrm{handler}:&\quad
 \forall\rho_B.\ \exists\rho_J.\ \rho_J\models D_o\land t_r\leq t_r\land\epsilon_o\subseteq\chi,\\
 &J\cap F_o=\varnothing,\quad B\cap K_o=\varnothing,
 \quad\Scope(B,J;H,\ldots).
\end{aligned}
\]
\end{proof}
\begin{theorem}[射影後の正確性]
\[
\Proj(\Gamma,e,t,\epsilon)=D\land\GReal(\Gamma,\rho)
\Longrightarrow
\bigl(\rho\models D\iff\ground\Gamma_\rho\vdash_g e:\eval t_\rho!\eval\epsilon_\rho\bigr).
\]
\end{theorem}
\begin{proof}
$(\Rightarrow)$：生成健全性と導出の意味解釈．
$(\Leftarrow)$：$\nu=_{F}\rho$ を前定理で得て，
$\omega=\rho[\supp D_{\mathrm{raw}}\setminus F\mapsto\nu]$ を存在証人とする．
\end{proof}
\begin{theorem}[成功時の健全性]
\[
\begin{gathered}
\Inf^{\mathrm{at}}_{N,S}(\Gamma,e,t,\epsilon)=\mathsf{solved}(R,\theta),\quad
\rho\models R,\quad\GReal(\Gamma,\theta^*\rho)\\
\Longrightarrow\quad
\ground\Gamma_{\theta^*\rho}\vdash_g e:\eval{\theta t}_\rho!\eval{\theta\epsilon}_\rho.
\end{gathered}
\]
\[
\Gamma'=\theta\Gamma,\quad\Real_\top(\Gamma),\quad\Adm_{\Sat}(\Gamma')
\quad\Longrightarrow\quad
\operatorname{Typing}_{\Sat}(R,\Gamma',e,\theta t,\theta\epsilon).
\]
\end{theorem}

最後の式は偽であり，未解決は検出能力の制限を表す．量化順序の交換は行わない．
\[
\begin{gathered}
\Sigma(o)=\forall\alpha.\alpha\to\alpha,\qquad
 e_h=\mathbf{handle}_o(\mathbf{perform}\ o\ 0_{\mathrm{int}},\;0,\;0\ 1),\\
[]\vdash_g e_h:\mathrm{Int}!\varnothing,
\qquad \Inf_{N,S_{\rm tc}}([],e_h)=\mathsf{solved}(R,\theta).
\end{gathered}
\]
